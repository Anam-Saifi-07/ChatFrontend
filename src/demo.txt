export const NavigationBar = ({ activeTab, setActiveTab, openCreateModal }) => {
  const tabs = [
    { id: 'person', label: 'Person', icon: '👤' },
    { id: 'profile', label: 'Profile', icon: '🆔' },
    { id: 'home', label: 'Home', icon: '🏠' },
    { id: 'room', label: 'Room', icon: '👥' },
  ];

  return (
    <div className="bg-white border-top p-2 d-flex justify-content-around align-items-center shadow-sm">
      {tabs.map((tab) => (
        <button
          key={tab.id}
          onClick={() => setActiveTab(tab.id)}
          className={`btn d-flex flex-column align-items-center btn-sm px-3 py-1 ${
            activeTab === tab.id ? 'btn-primary text-white shadow-sm' : 'btn-light text-secondary border-0'
          }`}
          style={{ minWidth: '70px', borderRadius: '10px' }}
        >
          <span className="fs-5">{tab.icon}</span>
          <small className="fw-semibold mt-1">{tab.label}</small>
        </button>
      ))}

      {/* Dynamic Create Button: Sirf 'Person' ya 'Room' tab par visible hoga */}
      {(activeTab === 'person' || activeTab === 'room') && (
        <button
          onClick={openCreateModal}
          className="btn btn-success rounded-circle d-flex align-items-center justify-content-center shadow"
          style={{ width: '45px', height: '45px', fontSize: '20px' }}
          title={`Create new ${activeTab}`}
        >
          +
        </button>
      )}
    </div>
  );
};
